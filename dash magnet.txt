-- Settings para sa Dash

local dashSpeed = 160 -- Mas gipaspasan pa para "pas pas" jud ang dasmag

local dashEnabled = true
local MAX_MAGNET_DISTANCE = 200 -- Mas layo nga lock-on distance

-- ==========================================
-- TARGETING / MAGNET DETECTION
-- ==========================================
local function getTargetUnderCursor()
	local targetPart = mouse.Target
	if targetPart and targetPart.Parent then
		local targetModel = targetPart.Parent
		local targetHumanoid = targetModel:FindFirstChildOfClass("Humanoid")
		local targetRoot = targetModel:FindFirstChild("HumanoidRootPart")

		if not targetHumanoid and targetModel.Parent then
			targetModel = targetModel.Parent
			targetHumanoid = targetModel:FindFirstChildOfClass("Humanoid")
			targetRoot = targetModel:FindFirstChild("HumanoidRootPart")
		end

		if targetHumanoid and targetRoot and targetModel ~= character and targetHumanoid.Health > 0 then
			local distance = (targetRoot.Position - rootPart.Position).Magnitude
			if distance <= MAX_MAGNET_DISTANCE then
				return targetRoot
			end
		end
	end

	-- Screen center search para bisag wala na point ang mouse but duol sa tunga sa screen
	local unitRay = camera:ScreenPointToRay(mouse.X, mouse.Y)
	local closestPlayerRoot = nil
	local closestDist = MAX_MAGNET_DISTANCE

	for _, otherPlayer in ipairs(Players:GetPlayers()) do
		if otherPlayer ~= player and otherPlayer.Character then
			local otherRoot = otherPlayer.Character:FindFirstChild("HumanoidRootPart")
			local otherHum = otherPlayer.Character:FindFirstChildOfClass("Humanoid")
			if otherRoot and otherHum and otherHum.Health > 0 then
				local toTarget = otherRoot.Position - unitRay.Origin
				local projectedDist = toTarget:Dot(unitRay.Direction.Unit)
				if projectedDist > 0 and projectedDist < closestDist then
					local closestPointOnRay = unitRay.Origin + (unitRay.Direction.Unit * projectedDist)
					local distFromRay = (otherRoot.Position - closestPointOnRay).Magnitude
					if distFromRay <= 15 then -- Daghang leeway para sayon ma-lock on
						closestDist = projectedDist
						closestPlayerRoot = otherRoot
					end
				end
			end
		end
	end

	return closestPlayerRoot
end

-- ==========================================
-- HIGH-SPEED HOMING / PILIT DASH MECHANIC
-- ==========================================
local function performDash()
	if not dashEnabled or humanoid.Health <= 0 then return end

	local targetRoot = getTargetUnderCursor()

	-- Limpyo sa karaang physical attributes (kung naa)
	local oldAttachment = rootPart:FindFirstChild("DashAttachment")
	if oldAttachment then oldAttachment:Destroy() end

	local attachment = Instance.new("Attachment")
	attachment.Name = "DashAttachment"
	attachment.Parent = rootPart

	if targetRoot then
		-- ACTIVE HOMING (MO-PILIT) - Mogamit og AlignPosition aron mo-gukod ug moliko real-time
		local targetAttachment = Instance.new("Attachment")
		targetAttachment.Name = "DashTarget"
		targetAttachment.Parent = targetRoot

		local alignPos = Instance.new("AlignPosition")
		alignPos.Mode = Enum.PositionAlignmentMode.TwoAttachment
		alignPos.Attachment0 = attachment
		alignPos.Attachment1 = targetAttachment
		alignPos.MaxForce = 3000000 -- Sobra ka kusog nga force para dili makalingkawas
		alignPos.MaxVelocity = dashSpeed
		alignPos.Responsiveness = 100 -- PINAKA-PASPAS NGA LIKO (Mura'g magnet)
		alignPos.Parent = attachment

		-- Dynamic Facing (Sigurohon nga naga-atubang ka sa imong gigukod samtang naglupad)
		local connection
		connection = RunService.Heartbeat:Connect(function()
			if targetRoot and targetRoot.Parent and rootPart and rootPart.Parent then
				rootPart.CFrame = CFrame.lookAt(rootPart.Position, targetRoot.Position)
			else
				if connection then connection:Disconnect() end
			end
		end)

		-- Limit Duration / Clean up
		task.delay(dashDuration, function()
			if connection then connection:Disconnect() end
			if attachment and attachment.Parent then attachment:Destroy() end
			if targetAttachment and targetAttachment.Parent then targetAttachment:Destroy() end
		end)

	else
		-- NORMAL DASH (Kung walay target, normal linear dash)
		local moveDir = humanoid.MoveDirection
		local dashDirection = (moveDir.Magnitude > 0) and moveDir.Unit or rootPart.CFrame.LookVector

		local linearVelocity = Instance.new("LinearVelocity")
		linearVelocity.MaxForce = 1000000
		linearVelocity.VectorVelocity = dashDirection * dashSpeed
		linearVelocity.Attachment0 = attachment
		linearVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
		linearVelocity.Parent = attachment

		task.delay(dashDuration, function()
			if attachment and attachment.Parent then
				attachment:Destroy()
			end
		end)
	end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == DASH_KEY then
		performDash()
	end
end)
