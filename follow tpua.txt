-- LOGIC VARIABLES
local activeTP = false
local partLimit = 50

-- FUNCTION TO GET UNANCHORED PARTS (WITH CAPACITY)
local function getTargetParts()
    local parts = {}
    local count = 0
    local limit = tonumber(CapacityInput.Text) or 50
    
    for _, part in pairs(workspace:GetDescendants()) do
        if count >= limit then break end
        if part:IsA("BasePart") and part.Anchored == false and not part:IsDescendantOf(LocalPlayer.Character) then
            local isOtherPlayer = false
            for _, p in pairs(Players:GetPlayers()) do
                if part:IsDescendantOf(p.Character or {}) then isOtherPlayer = true end
            end

            if not isOtherPlayer then
                table.insert(parts, part)
                count = count + 1
            end
        end
    end
    return parts
end

local function stopTP()
    activeTP = false
    Camera.CameraSubject = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
end

local function tpUnanchored(targetPlr)
    activeTP = true
    local selectedParts = getTargetParts()
    
    for _, part in pairs(selectedParts) do
        -- Clear old forces
        for _, c in pairs(part:GetChildren()) do
            if c:IsA("BodyPosition") or c:IsA("BodyAngularVelocity") or c:IsA("BodyVelocity") then c:Destroy() end
        end

        local bp = Instance.new("BodyPosition")
        bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bp.P = 50000; bp.D = 500; bp.Parent = part
        
        local bav = Instance.new("BodyAngularVelocity")
        bav.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
        bav.AngularVelocity = Vector3.new(150, 150, 150); bav.Parent = part
        
        task.spawn(function()
            while activeTP and bp.Parent and targetPlr.Character and targetPlr.Character:FindFirstChild("HumanoidRootPart") do
                bp.Position = targetPlr.Character.HumanoidRootPart.Position
                task.wait()
            end
            if bp.Parent then bp:Destroy() end
            if bav.Parent then bav:Destroy() end
        end)
    end
end

-- UPDATE LIST

    for _, plr in pairs(Players:GetPlayers()) do
